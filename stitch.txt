        {

          filename: "IMG_20261004_191244.jpg",
          alt: "Two travelers enjoying hot brewed filter coffee on a wooden ridge overlooking endless foggy pine hills bathed in early morning warm sunlight. Crisp authentic documentary photography, natural golden hour rim light, candid expressions, subtle steaming cup details."
        },
        {
          title: "Emerald Tea Terraces in Cloud Sea",
          subtitle: "Oct 3, 2026 • 5:50 PM • Madikeri Valley",
          filename: "IMG_20261003_175010.jpg",
          alt: "Sweeping landscape shot of geometric emerald tea plantations descending into white fluffy cloud formations during soft golden twilight. High dynamic range, vivid lush greens, cinematic dusk sky with lilac and apricot gradients."

      ];

      let currentIndex = 0;
      let currentRotation = 0;
      let isZoomed = false;
      let isFavorite = true;
      let isDrawerOpen = true;

      // Elements
      const activePhoto = document.getElementById('active-photo');
      const transformTarget = document.getElementById('transform-target');
      const photoTitle = document.getElementById('photo-title');
      const photoSubtitle = document.getElementById('photo-subtitle');
      const metaFilename = document.getElementById('meta-filename');

      const btnPrev = document.getElementById('btn-prev');
      const btnNext = document.getElementById('btn-next');
      const btnZoom = document.getElementById('btn-zoom');
      const zoomIcon = document.getElementById('zoom-icon');
      const btnInfo = document.getElementById('btn-info');
      const btnCloseDrawer = document.getElementById('btn-close-drawer');
      const infoDrawer = document.getElementById('info-drawer');
      const btnFavorite = document.getElementById('btn-favorite');
      const favIcon = document.getElementById('fav-icon');
      const btnMore = document.getElementById('btn-more');
      const contextMenu = document.getElementById('context-menu');
      const toast = document.getElementById('toast');
      const toastText = document.getElementById('toast-text');
      const toastIcon = document.getElementById('toast-icon');

      const menuRotate = document.getElementById('menu-rotate');
      const menuDownload = document.getElementById('menu-download');
      const menuSlideshow = document.getElementById('menu-slideshow');
      const menuArchive = document.getElementById('menu-archive');
      const menuTrash = document.getElementById('menu-trash');
      const btnTrash = document.getElementById('btn-trash');

      function showToast(msg, iconName = 'check_circle') {
        toastText.textContent = msg;
        toastIcon.textContent = iconName;
        toast.classList.remove('opacity-0', 'translate-y-2');
        toast.classList.add('opacity-100', 'translate-y-0');
        setTimeout(() => {
          toast.classList.add('opacity-0', 'translate-y-2');
          toast.classList.remove('opacity-100', 'translate-y-0');
        }, 2200);
      }

      function updatePhoto(idx) {
        currentIndex = (idx + photoLibrary.length) % photoLibrary.length;
        const p = photoLibrary[currentIndex];
        
        // Reset transforms
        currentRotation = 0;
        isZoomed = false;
        applyTransform();
        zoomIcon.textContent = 'zoom_in';

        // Update Text
        photoTitle.textContent = p.title;
        photoSubtitle.textContent = p.subtitle;
        metaFilename.textContent = p.filename;

        // Update image alt prompt
        activePhoto.setAttribute('data-alt', p.alt);
        activePhoto.alt = p.title;
      }

      function applyTransform() {
        const scale = isZoomed ? 1.5 : 1.0;
        transformTarget.style.transform = `scale(${scale}) rotate(${currentRotation}deg)`;
        activePhoto.style.cursor = isZoomed ? 'zoom-out' : 'zoom-in';
      }

      // Next / Previous
      btnPrev.addEventListener('click', () => updatePhoto(currentIndex - 1));
      btnNext.addEventListener('click', () => updatePhoto(currentIndex + 1));

      // Zoom Toggle
      function toggleZoom() {
        isZoomed = !isZoomed;
        zoomIcon.textContent = isZoomed ? 'zoom_out' : 'zoom_in';
        applyTransform();
      }
      btnZoom.addEventListener('click', toggleZoom);
      activePhoto.addEventListener('click', toggleZoom);

      // Rotate Left
      menuRotate.addEventListener('click', () => {
        currentRotation -= 90;
        applyTransform();
        showToast(`Rotated to ${((currentRotation % 360) + 360) % 360}°`, 'rotate_left');
        contextMenu.classList.add('hidden');
      });

      // Favorite Toggle
      btnFavorite.addEventListener('click', () => {
        isFavorite = !isFavorite;
        if (isFavorite) {
          favIcon.style.fontVariationSettings = "'FILL' 1";
          favIcon.classList.add('text-rose-400');
          showToast('Added to Favorites', 'favorite');
        } else {
          favIcon.style.fontVariationSettings = "'FILL' 0";
          favIcon.classList.remove('text-rose-400');
          showToast('Removed from Favorites', 'favorite_border');
        }
      });

      // Drawer Toggle
      function toggleDrawer() {
        isDrawerOpen = !isDrawerOpen;
        if (isDrawerOpen) {
          infoDrawer.classList.remove('hidden', '-mr-96');
          btnInfo.classList.remove('bg-white/10', 'text-white/90');
          btnInfo.classList.add('bg-primary', 'text-white', 'shadow-lg');
        } else {
          infoDrawer.classList.add('hidden', '-mr-96');
          btnInfo.classList.remove('bg-primary', 'text-white', 'shadow-lg');
          btnInfo.classList.add('bg-white/10', 'text-white/90');
        }
      }
      btnInfo.addEventListener('click', toggleDrawer);
      btnCloseDrawer.addEventListener('click', toggleDrawer);

      // Context Menu Toggle
      btnMore.addEventListener('click', (e) => {
        e.stopPropagation();
        contextMenu.classList.toggle('hidden');
      });

      document.addEventListener('click', (e) => {
        if (!contextMenu.contains(e.target) && e.target !== btnMore) {
          contextMenu.classList.add('hidden');
        }
      });

      // Menu Actions
      menuDownload.addEventListener('click', () => {
        showToast('Downloading original file (4.8 MB)...', 'download');
        contextMenu.classList.add('hidden');
      });

      menuSlideshow.addEventListener('click', () => {
        showToast('Starting slideshow...', 'slideshow');
        contextMenu.classList.add('hidden');
      });

      menuArchive.addEventListener('click', () => {
        showToast('Moved photo to Archive', 'archive');
        contextMenu.classList.add('hidden');
      });

      const handleDelete = () => {
        showToast('Moved to Trash. Undo?', 'delete');
        contextMenu.classList.add('hidden');
      };
      menuTrash.addEventListener('click', handleDelete);
      btnTrash.addEventListener('click', handleDelete);

      // Global Keyboard navigation
      window.addEventListener('keydown', (e) => {
        if (e.key === 'ArrowRight') {
          updatePhoto(currentIndex + 1);
        } else if (e.key === 'ArrowLeft') {
          updatePhoto(currentIndex - 1);
        } else if (e.key.toLowerCase() === 'i') {
          toggleDrawer();
        } else if (e.key.toLowerCase() === 'z') {
          toggleZoom();
        } else if (e.key === ' ' || e.code === 'Space') {
          e.preventDefault();
          showToast('Slideshow playback active', 'slideshow');
        } else if (e.key === 'Escape') {
          if (!contextMenu.classList.contains('hidden')) {
            contextMenu.classList.add('hidden');
          } else if (isDrawerOpen) {
            toggleDrawer();
          }
        }
      });
